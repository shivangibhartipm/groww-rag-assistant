#
# These mirror the environment that data/vector_index was built with. ChromaDB

# embedding model, or retrieval silently returns nothing useful.

# onnxruntime, so only tokenizers is genuinely new here.
chromadb==1.5.9
onnxruntime==1.28.0
tokenizers==0.22.2

# Fetches the model in scripts/fetch_model.py (build time, not per request)

# is deliberately absent.
requests>=2.32.2

# CA bundles for groww.in, HuggingFace and Groq
certifi==2026.4.22
